df = total_expense.copy()

# -----------------------------
# 7. METRIC FUNCTION
# -----------------------------
def calculate_metrics(actual, predicted):
    actual = np.array(actual)
    predicted = np.array(predicted)

    mae = mean_absolute_error(actual, predicted)
    rmse = np.sqrt(mean_squared_error(actual, predicted))

    non_zero_mask = actual != 0
    if np.sum(non_zero_mask) > 0:
        mape = np.mean(
            np.abs((actual[non_zero_mask] - predicted[non_zero_mask]) / actual[non_zero_mask])
        ) * 100
    else:
        mape = np.nan

    return mae, rmse, mape

# -----------------------------
# 9. LEAVE-ONE-OUT CROSS-VALIDATION
# -----------------------------
# LOOCV is ideal for tiny datasets because every observation matters.

loo = LeaveOneOut()

model_results = []

for model_name, model in models.items():
    actuals = []
    preds = []

    for train_idx, test_idx in loo.split(X):
        X_train, X_test = X.iloc[train_idx], X.iloc[test_idx]
        y_train, y_test = y.iloc[train_idx], y.iloc[test_idx]

        model.fit(X_train, y_train)
        y_pred = model.predict(X_test)[0]

        actuals.append(y_test.values[0])
        preds.append(y_pred)

    mae, rmse, mape = calculate_metrics(actuals, preds)

    result_row = {
        "Model": model_name,
        "MAE": mae,
        "RMSE": rmse,
        "MAPE (%)": mape
    }
    model_results.append(result_row)

comparison_df = pd.DataFrame(model_results).sort_values("RMSE").reset_index(drop=True)

print("\nModel comparison (LOOCV):")
print(comparison_df)
